# Native build adapter for the pinned Boost components used by Cycles/OIIO.
# No Boost.Build engine, bootstrap tool, or generated parser is imported.
cmake_minimum_required(VERSION 3.16)
project(CodeBrixBoost LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
find_package(Threads REQUIRED)
find_package(ZLIB REQUIRED)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT BOOST_SOURCE_ROOT)
  message(FATAL_ERROR "BOOST_SOURCE_ROOT must name the verified vendored snapshot")
endif()
function(boost_component name)
  set(sources)
  foreach(file IN LISTS ARGN)
    list(APPEND sources "${BOOST_SOURCE_ROOT}/libs/${name}/src/${file}")
  endforeach()
  add_library(boost_${name} SHARED ${sources})
  target_include_directories(boost_${name} PUBLIC
    "$<BUILD_INTERFACE:${BOOST_SOURCE_ROOT}>" "$<INSTALL_INTERFACE:include>")
  target_compile_definitions(boost_${name} PRIVATE BOOST_ALL_NO_LIB)
  if(WIN32)
    # Windows DLLs: Boost.Build defines BOOST_<LIB>_SOURCE while compiling a
    # library (dllexport instead of dllimport); consumers link the DLLs (dllimport)
    # and must not use Boost's auto-link pragmas, which name Boost.Build's files.
    string(TOUPPER "${name}" upper)
    target_compile_definitions(boost_${name} PRIVATE BOOST_${upper}_SOURCE PUBLIC BOOST_ALL_DYN_LINK BOOST_ALL_NO_LIB)
  endif()
  set_target_properties(boost_${name} PROPERTIES VERSION 1.83.0 SOVERSION 1.83.0 EXPORT_NAME ${name})
  install(TARGETS boost_${name} EXPORT CodeBrixBoostTargets LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
    ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endfunction()
file(GLOB fs RELATIVE "${BOOST_SOURCE_ROOT}/libs/filesystem/src" "${BOOST_SOURCE_ROOT}/libs/filesystem/src/*.cpp")
boost_component(filesystem ${fs})
target_compile_definitions(boost_filesystem PRIVATE BOOST_FILESYSTEM_DYN_LINK)
boost_component(chrono chrono.cpp thread_clock.cpp process_cpu_clocks.cpp)
target_compile_definitions(boost_chrono PRIVATE BOOST_CHRONO_DYN_LINK)
if(WIN32)
  boost_component(thread win32/thread.cpp win32/tss_dll.cpp win32/tss_pe.cpp future.cpp)
else()
  boost_component(thread pthread/thread.cpp pthread/once.cpp future.cpp)
endif()
target_compile_definitions(boost_thread PRIVATE BOOST_THREAD_BUILD_DLL)
target_link_libraries(boost_thread PRIVATE Threads::Threads boost_chrono)
boost_component(date_time gregorian/greg_month.cpp)
target_compile_definitions(boost_date_time PRIVATE BOOST_DATE_TIME_DYN_LINK)
file(GLOB regex_src RELATIVE "${BOOST_SOURCE_ROOT}/libs/regex/src" "${BOOST_SOURCE_ROOT}/libs/regex/src/*.cpp")
boost_component(regex ${regex_src})
target_compile_definitions(boost_regex PRIVATE BOOST_REGEX_DYN_LINK BOOST_REGEX_NO_ICU)
boost_component(iostreams file_descriptor.cpp mapped_file.cpp zlib.cpp gzip.cpp)
target_compile_definitions(boost_iostreams PRIVATE BOOST_IOSTREAMS_DYN_LINK)
target_link_libraries(boost_iostreams PRIVATE ZLIB::ZLIB)
# Boost.System has been header-only since 1.69. Retain the native compatibility
# library expected by the pinned Cycles FindBoost component list.
boost_component(system error_code.cpp)
install(DIRECTORY "${BOOST_SOURCE_ROOT}/boost" DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
set(config_dir "${CMAKE_INSTALL_LIBDIR}/cmake/Boost-1.83.0")
install(EXPORT CodeBrixBoostTargets FILE CodeBrixBoostTargets.cmake NAMESPACE Boost:: DESTINATION "${config_dir}")
configure_package_config_file(BoostConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/BoostConfig.cmake" INSTALL_DESTINATION "${config_dir}")
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/BoostConfigVersion.cmake" VERSION 1.83.0 COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/BoostConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/BoostConfigVersion.cmake" DESTINATION "${config_dir}")
